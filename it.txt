EXPERIMENT 2:
Q.	Create a program that illuminates the green LED if the counter is less than 100, illuminates the yellow LED if the counter is between 101 and 200 and illuminates the red LED if the counter is greater than 200.

CODE:
int counter = 0;
int redPin = 9;
int yellowPin = 10;
int greenPin = 11;
void setup() {
  pinMode(redPin, OUTPUT);
  pinMode(yellowPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
}
void loop() {
  counter++;
  if (counter < 101) {
    digitalWrite(greenPin, HIGH);
  } else if (counter < 201) {
    digitalWrite(yellowPin, HIGH);
  } else {
    digitalWrite(redPin, HIGH);
  }
  delay(10);
  digitalWrite(greenPin, LOW);
  digitalWrite(yellowPin, LOW);
  digitalWrite(redPin, LOW);
}

EXPERIMENT 3:
Q.Create a program so that when the user enters "B" the green light blinks, "G" the green light is illuminated "Y" the yellow light is illuminated and "R" the red light is illuminated.

CODE:
int redPin = 9;
int yellowPin = 10;
int greenPin = 11;
char input;
void setup() {
  pinMode(redPin, OUTPUT);
  pinMode(yellowPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  Serial.begin(9600);
}
void loop() {
  if (Serial.available() > 0) {
    input = Serial.read();
    if (input == 'B') {
      for (int i = 0; i < 5; i++) {
        digitalWrite(greenPin, HIGH);
        delay(100);
        digitalWrite(greenPin, LOW);
        delay(100);
      }
    } else if (input == 'G') {
      digitalWrite(greenPin, HIGH);
    } else if (input == 'Y') {
      digitalWrite(yellowPin, HIGH);
    } else if (input == 'R') {
      digitalWrite(redPin, HIGH);
    }
  }
}

EXPERIMENT 5:
Q.Write a program to control the color of the LED by turning potentiometer.

CODE:
const int R = 9;
const int G = 10;
const int B = 11;
const int RP = A0;
const int GP = A1;
const int BP = A2;
void setup() {
  pinMode(R, OUTPUT);
  pinMode(G, OUTPUT);
  pinMode(B, OUTPUT);
}
void loop() {
  analogWrite(R, analogRead(RP) / 4);
  analogWrite(G, analogRead(GP) / 4);
  analogWrite(B, analogRead(BP) / 4);

const int THERMISTOR_PIN = A0;
void setup() {
  Serial.begin(9600);
}
void loop() {
  float reading = analogRead(THERMISTOR_PIN);
  float voltage = reading * 5.0;
  voltage /= 1024.0;
  float temperature = (voltage - 0.5) * 100;

EXPERIMENT 7:
Q. Write a program so it displays the temperature in Fahrenheit as well as the maximum and minimum temperatures it has seen.

CODE:
#include <LiquidCrystal.h>
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);
const int tempPin = A0;
float temp, maxTemp = -100.0, minTemp = 100.0;
void setup() {
  lcd.begin(16, 2);
  lcd.print("Temp: ");
}

void loop() {
  temp = (5.0 * analogRead(tempPin) * 100.0) / 1024.0;
  temp = (temp - 500.0) / 10.0;
  lcd.setCursor(7, 0);
  lcd.print(temp);
  lcd.print(" F");
  if (temp > maxTemp) {
    maxTemp = temp;
    lcd.setCursor(0, 1);
    lcd.print("Max: ");
    lcd.print(maxTemp);
    lcd.print(" F");
  }
  if (temp < minTemp) {
    minTemp = temp;
    lcd.setCursor(0, 1);
    lcd.print("Min: ");
    lcd.print(minTemp);
    lcd.print(" F");
  }
  delay(1000);
}
